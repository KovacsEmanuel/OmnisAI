"""Parent-owned PTY issuer and private stdin transport for installed P1 QA."""
import base64
import json
import os
from pathlib import Path
import pty
import re
import select
import signal
import subprocess
import sys
import time

WORK = Path('/home/kali/.hermes/cache/scratch/p1-delivery-682449b-wNYBV6QY/reviewed')
stage = sys.argv[1]
if stage not in {'before', 'after'}:
    raise SystemExit(2)
master, slave = pty.openpty()
pair = subprocess.Popen(['/home/kali/.local/bin/omnis', 'pair'], cwd=WORK,
                        stdin=slave, stdout=slave, stderr=subprocess.PIPE,
                        start_new_session=True)
os.close(slave)
raw = bytearray()
try:
    deadline = time.monotonic() + 45
    while time.monotonic() < deadline:
        ready, _, _ = select.select([master], [], [], min(0.2, max(0, deadline-time.monotonic())))
        if ready:
            try:
                chunk = os.read(master, 4096)
            except OSError:
                break
            if not chunk:
                break
            raw.extend(chunk)
            if len(raw) > 4096:
                raise RuntimeError('PAIR_OUTPUT_LIMIT')
        if pair.poll() is not None and not ready:
            break
    pair.wait(timeout=max(0.1, deadline-time.monotonic()))
    assert pair.returncode == 0
    matches = re.findall(rb'Cod pairing Omnis \(expir\xc4\x83 \xc3\xaen 5 minute\): ([A-Za-z0-9_-]{32})\r?\n', bytes(raw))
    assert len(matches) == 1, 'PRIVATE_PAIRING_FORMAT'
    code = matches[0]
finally:
    os.close(master)
    if pair.poll() is None:
        os.killpg(pair.pid, signal.SIGTERM)
        pair.wait(timeout=5)
    if pair.stderr:
        pair.stderr.close()
    raw[:] = b'\x00' * len(raw)
    del raw
node = '/home/kali/.hermes/cache/scratch/omnis-p1-tools/node-prefix/node_modules/node-linux-x64/bin/node'
env = {'PATH': '/usr/bin:/bin', 'HOME': '/home/kali', 'TMPDIR': str(WORK),
       'PLAYWRIGHT_BROWSERS_PATH': '/home/kali/.cache/ms-playwright'}
driver = subprocess.Popen([node, str(WORK / 'p1-browser.mjs'), stage], cwd=WORK, env=env,
                          stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                          start_new_session=True)
try:
    out, err = driver.communicate(code + b'\n', timeout=310)
except subprocess.TimeoutExpired:
    os.killpg(driver.pid, signal.SIGTERM)
    try:
        driver.communicate(timeout=35)
    except subprocess.TimeoutExpired:
        os.killpg(driver.pid, signal.SIGKILL)
        driver.communicate(timeout=10)
    print('QA_TIMEOUT_UNCERTAIN')
    raise SystemExit(4)
variants = [code, base64.b64encode(code)]
assert not any(v in out or v in err for v in variants), 'SECRET_IN_DRIVER_OUTPUT'
# Compare the actual issued code against all actual runtime logs, without exporting it.
journal = json.loads(Path('/home/kali/.local/state/omnis-p1/r1/omnis/startup.json').read_text())
log_channels = []
for service, cid in journal['containers'].items():
    p = subprocess.run(['docker', 'logs', '--tail', '300', cid], capture_output=True, timeout=20)
    assert p.returncode == 0
    assert not any(v in p.stdout or v in p.stderr for v in variants), 'CODE_IN_RUNTIME_LOGS'
    log_channels.append(service)
forwarder = Path('/home/kali/.local/state/omnis-p1/r1/omnis/forwarder.log').read_bytes()
assert not any(v in forwarder for v in variants), 'CODE_IN_FORWARDER_LOG'
del code, variants, matches
report = json.loads((WORK / f'p1-{stage}-report.json').read_text())
receipt = json.loads((WORK / f'p1-{stage}-browser-close.json').read_text())
assert receipt['status'] == 'PASS', 'BROWSER_CLOSE_UNCERTAIN'
summary = {'stage': stage, 'driver_exit': driver.returncode, 'status': report['status'],
           'browser_close': receipt['status'], 'private_issued_code_log_scan': 'PASS',
           'log_channels': log_channels + ['forwarder'], 'checks': len(report['checks'])}
path = WORK / f'parent-{stage}.json'
path.write_text(json.dumps(summary, indent=2) + '\n'); path.chmod(0o600)
print(json.dumps(summary))
if driver.returncode or report['status'] != 'PASS':
    failed = [x for x in report['checks'] if x['status'] == 'FAIL']
    print(json.dumps({'failed_tags': failed}))
    raise SystemExit(1)
