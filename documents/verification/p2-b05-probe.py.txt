class Refusal(RuntimeError):
    pass


def require(condition, message):
    if not condition:
        raise Refusal(message)


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def load_file(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    require(spec is not None and spec.loader is not None, 'module_spec')
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module  # dataclasses resolve __module__ through sys.modules.
    spec.loader.exec_module(module)
    return module


def run(argv, env, timeout=20):
    result = subprocess.run(argv, env=env, capture_output=True, timeout=timeout,
                            stdin=subprocess.DEVNULL, check=False)
    require(result.returncode == 0, 'command_failed')
    require(len(result.stdout) + len(result.stderr) <= 16 * 1024 * 1024,
            'command_output_limit')
    return result.stdout.decode('utf-8')


def private(path, directory=False):
    info = path.lstat()
    require(info.st_uid == os.getuid() and stat.S_IMODE(info.st_mode) ==
            (0o700 if directory else 0o600), 'private_path')
    require(stat.S_ISDIR(info.st_mode) if directory else
            stat.S_ISREG(info.st_mode) and info.st_nlink == 1, 'private_path_type')


def control(directory, command):
    private(directory, True)
    private(directory / 'control.cap')
    info = (directory / 'control.sock').lstat()
    require(stat.S_ISSOCK(info.st_mode) and info.st_uid == os.getuid() and
            stat.S_IMODE(info.st_mode) == 0o600, 'control_socket')
    capability = (directory / 'control.cap').read_text()
    require(re.fullmatch('[0-9a-f]{64}', capability) is not None, 'control_capability')
    with socket.socket(socket.AF_UNIX) as channel:
        channel.settimeout(4)
        channel.connect(str(directory / 'control.sock'))
        import struct
        peer = struct.unpack('3i', channel.getsockopt(socket.SOL_SOCKET, socket.SO_PEERCRED, 12))
        require(peer[1] == os.getuid(), 'control_peer')
        channel.sendall(json.dumps({'version': 1, 'capability': capability,
                                   'command': command}).encode() + b'\n')
        with channel.makefile('rb') as stream:
            raw = stream.readline(8193)
        require(len(raw) <= 8192 and raw.endswith(b'\n'), 'control_frame')
        reply = json.loads(raw)
        require(type(reply) is dict and reply.get('ok') is True, 'control_refused')
        require(str(__import__('uuid').UUID(reply['correlation_id'])) == reply['correlation_id'],
                'control_correlation')
        return reply


def poll_until(fn, seconds, message):
    deadline = time.monotonic() + seconds
    while time.monotonic() < deadline:
        result = fn()
        if result:
            return result
        time.sleep(0.05)
    raise Refusal(message)


def write_report(path, report):
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(fd, 'w') as file:
        json.dump(report, file, indent=2, sort_keys=True)
        file.write('\n')
        file.flush()
        os.fsync(file.fileno())

            "print(json.dumps(seen))\n")
        sensitivity = json.loads(run([*docker, 'exec', '--user', f'{os.getuid()}:{os.getgid()}',
            '--workdir', '/state/work', rt_cid, '/opt/hermes/venv/bin/python', '-c', sensitivity_code], base))
        require(sensitivity == {'tui_gateway': True, 'hermes_bootstrap': True, 'markers_removed': True},
                'cwd_shadow_sensitivity_failed')
        shadow_readback('cwd-after-negative')
        rows, _ = top()
        require(len(rows) == 1, 'sensitivity_process_not_joined')
        report['cwd_shadow_control'] = {'shadow_hashes': shadow_hashes,
            'installed_nonisolated_sensitivity': sensitivity,
            'pre_native_marker_absence': True,
            'scope': 'Synthetic imports only on exact installed image; no native mutation or product edit.'}
        # Stock installed supervisor owns the only runtime connection.
        supervisor = subprocess.Popen([sys.executable, '-I', '-m', 'omnis.supervisor'],
            cwd=root, env={**migration_env, 'OMNIS_CONTROL_DIR': str(control_dir),
                           'OMNIS_RUNTIME_DIR': str(runtime), 'OMNIS_RUNTIME_INERT_PROBE': '1'},
            stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
            start_new_session=True)
        supervisor_fd = os.pidfd_open(supervisor.pid)
        require(select.select([supervisor.stdout], [], [], 8)[0], 'supervisor_start_deadline')
        require(supervisor.stdout.readline() == b'SUPERVISOR_READY\n', 'supervisor_ready_marker')
        initial = control(control_dir, {'op': 'status'})['status']
        require(initial['generation'] == 0 and initial['supervisor_epoch'] == 1 and
                initial['supervisor_ready'] is True and initial['admission_open'] is False and
                initial['assessment_ready'] is False and initial['shutdown_command_id'] is None,
                'inert_supervisor_state')
        rows, _ = top()
        native = [r for r in rows if r[2] == '/opt/hermes/venv/bin/python -I -m tui_gateway.entry']
        require(len(rows) == 2 and len(native) == 1 and int(native[0][1]) == owner_pid, 'real_native_process')
        native_pid = int(native[0][0])
        report['runtime']['native_host_pid'] = native_pid
        native_fds.append(os.pidfd_open(native_pid))
        checks['stock_NativePeer_handshake_and_unknown_refusal'] = True
        started = time.monotonic()
        lease_samples = []
        previous_expiry = None
        while time.monotonic() - started < 5.5:
            require(supervisor.poll() is None and not select.select(native_fds, [], [], 0)[0], 'continuity_lost')
            row, lease = snapshot()
            require(row[1] == 0 and row[2] is False and row[3] is None and lease[0] is not None and
                    lease[1] == 1 and lease[2] > lease[3], 'current_lease')
            if previous_expiry != lease[2]:
                lease_samples.append({'at_seconds': round(time.monotonic() - started, 3),
                                      'expires_at': lease[2].isoformat()})
                previous_expiry = lease[2]
            time.sleep(0.15)
        require(len(lease_samples) >= 3 and lease_samples[-1]['at_seconds'] - lease_samples[0]['at_seconds'] >= 2,
                'two_renew_intervals_not_observed')
        rows, _ = top()
        require(len(rows) == 2 and any(int(r[0]) == native_pid for r in rows), 'native_not_continuous')
        shadow_readback('cwd-after-real-native')
        report['cwd_shadow_control']['real_isolated_native_marker_absence'] = True
        checks['installed_cwd_shadow_isolation_and_sensitive_negative'] = True
        report['lease_samples'] = lease_samples
        checks['native_continuity_over_5_5_seconds_and_multiple_owner_renewals'] = True
        shutdown_id, finalize_id = uuid4(), uuid4()
        shutdown = control(control_dir, {'op': 'shutdown', 'instance_id': initial['instance_id'],
                            'command_id': str(shutdown_id), 'expected_generation': 0})
        require(shutdown['command_id'] == str(shutdown_id) and shutdown['expected_generation'] == 0 and
                shutdown['owner_epoch'] == 1 and shutdown['result_generation'] == 1 and
                shutdown['status']['generation'] == 1 and shutdown['status']['admission_open'] is False and
                shutdown['status']['shutdown_command_id'] == str(shutdown_id), 'SHUTDOWN_receipt')
        def quiesced():
            view = control(control_dir, {'op': 'status'})['status']
            return view if view['quiesced_generation'] == 1 else None
        poll_until(quiesced, 5, 'quiescence_missing')
        rows, _ = top()
        require(len(rows) == 1 and select.select(native_fds, [], [], 0)[0] == native_fds, 'native_family_not_drained')
        row, lease = snapshot()
        require(row[1] == 1 and row[2] is False and row[3] == shutdown_id and
                row[4] == 1 and row[5] is not None, 'durable_quiescence')
        final = control(control_dir, {'op': 'finalize', 'instance_id': initial['instance_id'],
                         'command_id': str(finalize_id), 'expected_generation': 1})
        require(final['command_id'] == str(finalize_id) and final['expected_generation'] == 1 and
                final['owner_epoch'] == 1 and final['result_generation'] == 1 and
                final['status']['shutdown_finalized'] is True and
                final['status']['quiesced_generation'] == 1, 'FINALIZE_receipt')
        with connect() as db:
            receipts = db.execute('SELECT command_id,kind,expected_generation,result_generation,owner_epoch '
                                  'FROM instance_command ORDER BY expected_generation').fetchall()
        require(receipts == [(shutdown_id, 'SHUTDOWN', 0, 1, 1), (finalize_id, 'FINALIZE', 1, 1, 1)], 'durable_command_readback')
        report['durable_commands'] = [{'command_id': str(r[0]), 'kind': r[1], 'expected_generation': r[2],
                                       'result_generation': r[3], 'owner_epoch': r[4]} for r in receipts]
        checks['UDS_SHUTDOWN_quiescence_FINALIZE_durable_readback'] = True
        checks['native_namespace_family_drained_before_finalize'] = True
        signal.pidfd_send_signal(supervisor_fd, signal.SIGTERM)
        out, err = supervisor.communicate(timeout=8)
        require(supervisor.returncode == 0 and out == err == b'', 'supervisor_normal_join')
        row, lease = snapshot()
        require(lease[:3] == (None, 1, None) and row[1] == 1 and row[4] == 1, 'supervisor_release_readback')
        require(not (control_dir / 'control.sock').exists() and not (control_dir / 'control.cap').exists(), 'control_cleanup')
        checks['supervisor_exit_zero_release_readback'] = True
        require({p: digest(REPO / 'api/src/omnis' / p) for p in sources} == api_hashes,
                'API_snapshot_changed_during_probe')
        require(all(digest(REPO / p) == value for p, value in installed_hashes.items()),
                'adapter_snapshot_changed_during_probe')
        require(run(['git', '-C', str(REPO), 'rev-parse', 'HEAD'], base).strip() == report['code_commit'],
                'HEAD_changed_during_probe')
        normal = True
    except BaseException as exc:
        report['failure'] = {'class': type(exc).__name__, 'reason': str(exc) if isinstance(exc, Refusal) else 'sanitized_failure'}
    finally:

        if supervisor is not None and supervisor.poll() is None:
            try:
                signal.pidfd_send_signal(supervisor_fd, signal.SIGTERM)
                supervisor.communicate(timeout=8)
            except BaseException:
                normal = False
                try:
                    signal.pidfd_send_signal(supervisor_fd, signal.SIGKILL)
                    supervisor.communicate(timeout=3)
                except BaseException:
                    family_clean = False
        if supervisor_fd is not None:
            os.close(supervisor_fd)
        cleanup['supervisor_joined'] = supervisor is None or supervisor.returncode is not None
        if rt_cid is None and docker is not None and root is not None:
            try:
                candidate = json.loads(run([*docker, 'inspect', runtime_name], base))[0]
                rt_cid = candidate['Id']
            except BaseException:
                pass
        if rt_cid is not None:
            try:
                runtime_attest()
                run([*docker, 'stop', '--time', '5', rt_cid], base, 15)
                stopped = runtime_attest()
                require(stopped['State']['Running'] is False and stopped['State']['Pid'] == 0, 'runtime_stop_readback')
                run([*docker, 'rm', rt_cid], base)
                daemon()
                absent = subprocess.run([*docker, 'inspect', rt_cid], env=base, capture_output=True, timeout=10)
                require(absent.returncode != 0 and f'No such object: {rt_cid}'.encode() in absent.stderr, 'runtime_absence')
                require(not run([*docker, 'ps', '-a', '--no-trunc', '--filter',
                                 'label=org.omnis.b05.probe=' + runtime_label, '--format', '{{.ID}}'], base).strip(), 'runtime_label_absence')
                require(not (runtime / 'runtime.sock').exists() and not (runtime / 'runtime.boot').exists(), 'runtime_UDS_cleanup')
                cleanup['runtime_exact_removed_socket_absent'] = True
            except BaseException:
                cleanup['runtime_exact_removed_socket_absent'] = False
                family_clean = False
        for fd in native_fds:
            if not select.select([fd], [], [], 0)[0]:
                family_clean = False
            os.close(fd)
        cleanup['native_family_settled'] = family_clean
        if paths is not None and docker is not None and project is not None:
            try:
                if db_cid is None:
                    matches = gate._find_owned(docker, base, project)
                    require(len(matches) <= 1, 'unresolved_DB_candidates')
                    db_cid = matches[0] if matches else None
                if family_clean and db_cid is not None:
                    if db_name is not None and conftest is not None and url is not None:
                        gate.attest_fixture(paths.root / 'fixture.json')
                        from psycopg import sql
                        with conftest.connect_test_database(url.set(database='omnis_test'), autocommit=True) as db:
                            db.execute(sql.SQL('DROP DATABASE IF EXISTS {}').format(sql.Identifier(db_name)))
                            require(db.execute('SELECT 1 FROM pg_database WHERE datname=%s', (db_name,)).fetchone() is None,
                                    'owned_schema_drop_readback')
                        cleanup['owned_database_absent'] = True
                    cleanup['DB_exact_removed'] = gate.cleanup_owned(db_cid, docker, base, paths, project, db_image, daemon_id)
                elif db_cid is not None:
                    cleanup['DB_retained_due_unresolved_family'] = True
                else:
                    cleanup['DB_not_created'] = not gate._find_owned(docker, base, project)
            except BaseException:
                cleanup['DB_exact_removed'] = False
            cleanup['fixture_data_retained'] = str(paths.root)
        required_cleanup = (cleanup.get('supervisor_joined') is True and family_clean and
                            cleanup.get('runtime_exact_removed_socket_absent') is True and
                            cleanup.get('owned_database_absent') is True and cleanup.get('DB_exact_removed') is True)
        report['verdict'] = 'PASS_SCOPED' if normal and required_cleanup else 'NOT_PASS'
        os.environ.clear()
        os.environ.update(old_env)
        write_report(report_path, report)
    print(json.dumps({'verdict': report['verdict'], 'report': str(report_path)}))
    return 0 if report['verdict'] == 'PASS_SCOPED' else 2

    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument('--image', required=True, help='Already installed immutable OCI sha256: ID; never pulls/builds.')
    parser.add_argument('--report', required=True, type=Path, help='New absolute JSON path under $TMPDIR (0600).')
    args = parser.parse_args()
    if re.fullmatch(r'sha256:[0-9a-f]{64}', args.image) is None:
        parser.error('--image must be an exact lowercase sha256: image ID')
    previous_term = signal.getsignal(signal.SIGTERM)

    def interrupt(_signum, _frame):
        raise KeyboardInterrupt

    signal.signal(signal.SIGTERM, interrupt)
    try:
        return probe(args)
    except BaseException as exc:
        print('B05_PROBE_BLOCKED:' + type(exc).__name__, file=sys.stderr)
        return 2
    finally:
        signal.signal(signal.SIGTERM, previous_term)
