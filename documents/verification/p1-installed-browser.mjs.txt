async function readCode() {
  must(!process.stdin.isTTY); // Parent pipes the private PTY capture; never interactive echo.
  return bounded(new Promise((resolve, reject) => {
    let chunks = [], length = 0;
    const clean = () => { process.stdin.removeListener('data', data); process.stdin.removeListener('end', end); process.stdin.removeListener('error', error); };
    const error = () => { clean(); reject(new Error('SANITIZED_STDIN')); };
    const data = chunk => { length += chunk.length; if (length > 130) { chunks = []; error(); process.stdin.pause(); } else chunks.push(chunk); };
    const end = () => {
      clean(); let value = Buffer.concat(chunks).toString('utf8'); chunks.forEach(chunk => chunk.fill(0)); chunks = [];
      // Accept one framing newline, never normalize whitespace within a token.
      value = value.replace(/\r?\n$/, '');
      if (!/^[A-Za-z0-9_-]{16,128}$/.test(value)) { reject(new Error('SANITIZED_CODE_FORMAT')); return; }
      resolve(value);
    };
    process.stdin.on('data', data); process.stdin.once('end', end); process.stdin.once('error', error); process.stdin.resume();
  }), 15000);
}

async function run() {
  tag = 'preflight'; must(process.argv.length === 3 && ['before', 'after'].includes(stage));
  must(dirname(fileURLToPath(import.meta.url)) === ROOT);
  must(JSON.parse(await readFile(join(PW, 'package.json'), 'utf8')).version === '1.63.0');
  must(JSON.parse(await readFile(join(TEST, 'package.json'), 'utf8')).version === '1.63.0');
  if (stage === 'before') {
    try { await lstat(recordPath); fail(); } catch (error) { must(error?.code === 'ENOENT'); }
  } else {
    record = JSON.parse(await readFile(recordPath, 'utf8'));
    must(record.schema === 1 && record.source_head === HEAD && record.origin === ORIGIN && record.before_logout_denials === true);
    must(record.authoritative_response.engagement_id === record.engagement_id && record.authoritative_response.revision === 3);
    must(record.authoritative_response.payload.objective_text === FINAL_OBJECTIVE);
  }
  const { chromium } = await import(`${PW}/index.mjs`);
  ({ expect } = await import(`${TEST}/index.mjs`)); expect = expect.configure({ timeout: 12000 });
  await mkdir(join(ROOT, 'browser-home'), { recursive: true, mode: 0o700 });
  tag = 'browser-launch';
  server = await chromium.launchServer({ headless: true, timeout: 20000, env: { PATH: '/usr/bin:/bin', HOME: join(ROOT, 'browser-home'), TMPDIR: ROOT } }).then(async owned => {
    server = owned;
    if (closing) { await bounded(owned.kill(), 10000); fail(); }
    return owned;
  });
  must(!closing);
  browser = await chromium.connect(server.wsEndpoint(), { timeout: 12000 });
  context = await browser.newContext({ viewport: { width: 390, height: 1000 }, serviceWorkers: 'block', acceptDownloads: false });
  context.setDefaultTimeout(12000); context.setDefaultNavigationTimeout(15000);
  await context.route('**/*', async route => {
    const url = route.request().url(); privateObserve(url);
    if (closing) { await route.abort(); return; }
    if (new URL(url).origin !== ORIGIN) { event('network-nonloopback-blocked', 'FAIL', '[non-loopback]'); await route.abort(); return; }
    await route.continue();
  });
  context.on('page', target => {
    target.on('console', message => {
      privateObserve(message.text());
      if (message.type() === 'error') {
        const expected = /^Failed to load resource: the server responded with a status of (400|401|403|409|422)\b/.test(message.text());
        event(expected ? 'console-expected-negative' : 'console-error', expected ? 'EXPECTED' : 'FAIL', safePath(message.location().url ?? ''));
      } else event('console-' + (['log', 'warning', 'info', 'debug'].includes(message.type()) ? message.type() : 'other'), 'OBSERVED');
    });
    target.on('pageerror', error => { privateObserve(error.message); event('pageerror', 'FAIL'); });
  });
  context.on('response', response => {
    privateObserve(response.url());
    if (response.status() >= 400) event('network-http', response.status(), safePath(response.url()));
    if (response.url() === `${ORIGIN}/api/session` || response.url() === `${ORIGIN}/api/pairing`) track((async () => {
      if (response.ok()) { const body = await response.json(); secret(body.csrf_token); }
    })());
  });
  context.on('requestfailed', request => {
    privateObserve(request.url()); privateObserve(request.failure()?.errorText ?? '');
    if (!closing && request.failure()?.errorText !== 'net::ERR_ABORTED') event('network-requestfailed', 'FAIL', safePath(request.url()));
  });
  page = await context.newPage();
  await step('private-cli-code-pair-readback', pair);
  await step('n01-truthful-installed-dashboard', async () => {
    await expect(page.getByText('Create and edit saved drafts. Assessment execution is not available in this build.', { exact: true })).toBeVisible();
    await expect(page.getByText(/Setup editing remains unavailable in B03/)).toHaveCount(0);
  });
  if (stage === 'before') await before(); else await after();
  await step('private-url-storage-console-leak-scan', inspectSecrets);
  must(!diagnostics.some(item => item.status === 'FAIL'));
  must(!diagnostics.some(item => item.tag === 'network-http' && ![400, 401, 403, 409, 422].includes(item.status)));
}
let cleanupStatus = 'NOT_STARTED', timer, signalReject;
const stopped = new Promise((_, reject) => { signalReject = reject; });
const onSignal = () => signalReject(new Error('SANITIZED_SIGNAL'));
const onUnexpected = () => { event('unexpected-runtime-error', 'FAIL'); signalReject(new Error('SANITIZED_RUNTIME')); };
process.once('SIGINT', onSignal); process.once('SIGTERM', onSignal);
process.on('unhandledRejection', onUnexpected); process.on('uncaughtException', onUnexpected);
let task;
try {
  timer = setTimeout(() => signalReject(new Error('SANITIZED_STAGE_TIMEOUT')), 240000);
  task = run(); await Promise.race([task, stopped]); verdict = 'PASS';
} catch (error) {
  checks.push({ tag, status: 'FAIL', path: null });
  if (page && !page.isClosed()) {
    try {
      await jsonFile(join(ROOT, 'p1-failure-dom.json'), {
        tag, error_type: error?.constructor?.name ?? 'Unknown',
        dom: await page.evaluate(() => ({
          active: { tag: document.activeElement?.tagName, id: document.activeElement?.id, role: document.activeElement?.getAttribute('role') },
          alerts: [...document.querySelectorAll('[role="alert"]')].map(e => ({ tag: e.tagName, heading: e.querySelector('h3')?.textContent, focused: document.activeElement === e })),
        })),
      });
    } catch { event('failure-dom-unavailable', 'OBSERVED'); }
  }
}
finally {
  clearTimeout(timer); closing = true;
  try {
    if (context) await bounded(context.close(), 10000);
    if (browser) await bounded(browser.close(), 10000);
    if (server) await bounded(server.close(), 10000);
    const owned = server?.process();
    if (owned) must(owned.exitCode !== null || owned.signalCode !== null);
    cleanupStatus = server ? 'PASS' : 'NOT_STARTED';
  } catch {
    try {
      if (server) await bounded(server.kill(), 10000);
      const owned = server?.process(); must(owned && (owned.exitCode !== null || owned.signalCode !== null));
      cleanupStatus = 'PASS_RESCUED';
    } catch { cleanupStatus = 'UNCERTAIN'; }
    verdict = 'FAIL';
  }
  // Settle any still-running owned task after closing its browser; never export error objects.
  if (task) { try { await bounded(task.catch(() => {}), 15000); } catch { cleanupStatus = 'UNCERTAIN'; verdict = 'FAIL'; } }
  try {
    for (const text of privateTexts) for (const value of knownSecrets) if (text.includes(value)) leaked = true;
    if (leaked || overflow) verdict = 'FAIL';
    if (verdict === 'PASS' && stage === 'before') await jsonFile(recordPath, record, true);
    await jsonFile(receiptPath, { tag: 'owned-browser-close', status: cleanupStatus, path: null });
    await jsonFile(reportPath, { tag: `installed-p1-${stage === 'after' ? 'after' : 'before'}`, status: verdict, path: recordPath,
      checks, diagnostics, browser_close: { tag: 'owned-browser-close', status: cleanupStatus, path: receiptPath },
      secret_scan: { tag: 'url-storage-console-private-scan', status: leaked || overflow ? 'FAIL' : 'PASS', path: null },
      limits: [
        { tag: 'actual-runtime-execution-only-not-source-fixture', status: 'SCOPED', path: ORIGIN },
        { tag: 'stop-start-db-logs-resource-attestation-parent-owned', status: 'PARENT_REQUIRED', path: null },
        { tag: 'before-logout-unchanged-proved-by-after-stage', status: stage === 'after' && verdict === 'PASS' ? 'PASS' : 'PENDING', path: recordPath },
        { tag: 'after-final-logout-db-unchanged-readback', status: 'PARENT_REQUIRED', path: null },
        { tag: 'no-request-bodies-or-secret-headers-exported-record-is-approved-nonsecret-get-response', status: 'SCOPED', path: recordPath },
        { tag: 'no-screen-reader-wcag-or-aesthetic-approval', status: 'NOT_RUN', path: null },
        { tag: 'reticle-not-integrated-chisel-no-go', status: 'NOT_RUN', path: null },
        { tag: 'runtime-daemon-logs-private-secret-scan', status: 'PARENT_REQUIRED', path: null },
      ] });
  } catch { verdict = 'FAIL'; }
  privateTexts = []; knownSecrets.clear();
  process.removeListener('SIGINT', onSignal); process.removeListener('SIGTERM', onSignal); process.stdin.pause();
  // No raw exception, code, cookie, CSRF, request body, URL query, browser WS endpoint or storage output.
  process.stdout.write(JSON.stringify({ tag: 'installed-p1-browser', status: verdict, path: reportPath }) + '\n');
  process.exit(verdict === 'PASS' && ['PASS', 'PASS_RESCUED'].includes(cleanupStatus) ? 0 : 1);
}
